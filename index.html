7

Create SKILL.md

@'
---
name: critical-support-escalation
description: Investigate critical ServiceNow incidents and prepare evidence-backed coordinated actions.
---

# Critical Support Escalation

1. Retrieve the relevant critical incidents.
2. Compare symptoms, affected services, priorities, states, and notes.
3. Group incidents only when the evidence supports a relationship.
4. Keep uncertain incidents separate.
5. Search for relevant Known Error resolutions.
6. Recommend a coordinated action supported by evidence.
7. Do not add a work note or modify an incident without explicit user confirmation.
8. Do not invent incident details.
'@ | Set-Content -Path .\skills\critical-support-escalation\SKILL.md -Encoding utf8
Open expected response

PowerShell normally returns to the prompt without displaying output. The file is created at:

skills\critical-support-escalation\SKILL.md

Create servicenow-critical-tools.json

Important: These pinned tool names and schemas must match the tools exposed by the instructor's MCP server.
@'
{
  "tools": [
    {
      "name": "get_critical_incidents",
      "description": "Retrieve critical ServiceNow incidents for investigation.",
      "inputSchema": {"type": "object", "properties": {}, "additionalProperties": false}
    },
    {
      "name": "get_incident",
      "description": "Retrieve the details of a ServiceNow incident by incident number.",
      "inputSchema": {
        "type": "object",
        "properties": {"incident_number": {"type": "string", "description": "ServiceNow incident number such as INC0010015."}},
        "required": ["incident_number"],
        "additionalProperties": false
      }
    },
    {
      "name": "search_known_errors",
      "description": "Search Known Error information for evidence-backed resolutions.",
      "inputSchema": {
        "type": "object",
        "properties": {"query": {"type": "string", "description": "Symptoms, service, or issue keywords."}},
        "required": ["query"],
        "additionalProperties": false
      }
    },
    {
      "name": "add_work_note",
      "description": "Add a confirmed coordinated work note to a ServiceNow incident.",
      "inputSchema": {
        "type": "object",
        "properties": {
          "incident_number": {"type": "string"},
          "work_note": {"type": "string"}
        },
        "required": ["incident_number", "work_note"],
        "additionalProperties": false
      }
    }
  ]
}
'@ | Set-Content -Path .\tools\servicenow-critical-tools.json -Encoding utf8
9

Create manifest.json

Set your instructor-provided MCP URL

$env:LAB_MCP_URL = "<YOUR-GROUP-DEV-TUNNEL-URL>/mcp"
Check: The URL must end with /mcp.

Generate an app ID

$appId = [guid]::NewGuid().ToString()

Create the manifest

$manifest = @{
  '$schema' = 'https://developer.microsoft.com/en-us/json-schemas/teams/vDevPreview/MicrosoftTeams.schema.json'
  manifestVersion = 'devPreview'
  version = '2.0.0'
  id = $appId
  developer = @{
    name = 'Critical Support Escalation Demo'
    websiteUrl = $env:LAB_MCP_URL -replace '/mcp$',''
    privacyUrl = $env:LAB_MCP_URL -replace '/mcp$',''
    termsOfUseUrl = $env:LAB_MCP_URL -replace '/mcp$',''
  }
  icons = @{color='color.png'; outline='outline.png'}
  name = @{short='Critical Support'; full='Critical Support Escalation for Copilot Cowork'}
  description = @{
    short='Investigate and act on multiple critical ServiceNow tickets.'
    full='A Cowork plugin for critical support escalation teams that reasons across multiple ServiceNow incidents, searches Known Error resolutions, recommends evidence-backed coordinated actions, and can add confirmed work notes.'
  }
  accentColor = '#D83B01'
  permissions = @('identity','messageTeamMembers')
  validDomains = @()
  agentConnectors = @(@{
    id='servicenow-critical-support'
    displayName='ServiceNow Critical Support'
    description='Retrieve critical incidents, inspect individual tickets, search Known Error resolutions, and add confirmed work notes in ServiceNow.'
    toolSource=@{remoteMcpServer=@{
      mcpServerUrl=$env:LAB_MCP_URL
      mcpToolDescription=@{file='tools/servicenow-critical-tools.json'}
      authorization=@{type='None'}
    }}
  })
  agentSkills = @(@{folder='skills/critical-support-escalation'})
}

$manifest | ConvertTo-Json -Depth 20 | Set-Content -Path .\manifest.json -Encoding utf8
